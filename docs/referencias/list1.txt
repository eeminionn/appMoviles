
import 'package:flutter/material.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import '../theme/theme.dart';

class ListsView extends StatefulWidget {
  const ListsView({super.key});

  @override
  State<ListsView> createState() => _ListsViewState();
}

class _ListsViewState extends State<ListsView> {
  // colección de firestore
  final canciones = FirebaseFirestore.instance.collection('canciones');

  // controladores de texto
  final tituloController = TextEditingController(); // cada elemento que se desse guardar debe llevar un controlador
  final subtituloController = TextEditingController();

  // validación del formulario
  final formKey = GlobalKey<FormState>();

  // estado del guardado
  bool guardando = false;

  @override
  void dispose() {
    tituloController.dispose();
    subtituloController.dispose();
    super.dispose();
  }

  // abrir formulario en bottom sheety, acá esta la función de guardar datos
  Future<void> agregarCancion() async { 
    tituloController.clear();
    subtituloController.clear();
    guardando = false;

    await showModalBottomSheet<void>(
      context: context,
      isScrollControlled: true,
      showDragHandle: true,
      useSafeArea: true,
      isDismissible: !guardando,
      builder: (sheetContext) {
        return StatefulBuilder(
          builder: (context, setSheetState) {
            // guardar canción en firestore
            Future<void> guardarCancion() async {
              if (!formKey.currentState!.validate()) {
                return;
              }

              setSheetState(() {
                guardando = true;
              });

              try {
                await canciones.add({
                  'titulo': tituloController.text.trim(), //correlativo a los controladores de arriba
                  'subtitulo': subtituloController.text.trim(),
                  'creadoEn': FieldValue.serverTimestamp(), //guarda la fecha del registro
                });

                if (!sheetContext.mounted) return;

                Navigator.pop(sheetContext); //levanta temporalmente el bottom sheet

                if (!mounted) return;

                ScaffoldMessenger.of(this.context).showSnackBar(
                  const SnackBar(
                    content: Text('Canción guardada correctamente'), //confirmación
                  ),
                );
              } catch (e) {
                if (!mounted) return;

                ScaffoldMessenger.of(this.context).showSnackBar(
                  SnackBar(
                    content: Text('Error al guardar: $e'),
                  ),
                );
              } finally {
                guardando = false;

                if (sheetContext.mounted) {
                  setSheetState(() {});
                }
              }
            }

            return Padding(
              padding: EdgeInsets.fromLTRB(
                AppSpacing.spacing24,
                AppSpacing.spacing8,
                AppSpacing.spacing24,
                MediaQuery.viewInsetsOf(context).bottom +
                    AppSpacing.spacing24,
              ),

              child: SingleChildScrollView(
                child: Form(
                  key: formKey,
                  child: Column(
                    mainAxisSize: MainAxisSize.min,
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      // título del formulario
                      Text(
                        'Nueva canción',
                        style: Theme.of(context).textTheme.headlineSmall,
                      ),

                      const SizedBox(
                        height: AppSpacing.spacing24,
                      ),

                      // campo título
                      TextFormField(
                        controller: tituloController,
                        decoration: const InputDecoration(
                          labelText: 'Título',
                          border: OutlineInputBorder(),
                        ),
                        validator: (value) {
                          if (value == null || value.trim().isEmpty) {
                            return 'Ingresa un título';
                          }
                          return null;
                        },
                      ),

                      const SizedBox(
                        height: AppSpacing.spacing16,
                      ),

                      // campo subtítulo
                      TextFormField(
                        controller: subtituloController,
                        decoration: const InputDecoration(
                          labelText: 'Subtítulo',
                          border: OutlineInputBorder(),
                        ),
                      ),

                      const SizedBox(
                        height: AppSpacing.spacing24,
                      ),

                      // botones del formulario
                      Row(
                        mainAxisAlignment: MainAxisAlignment.end,
                        children: [
                          TextButton(
                            onPressed: guardando
                                ? null
                                : () => Navigator.pop(sheetContext),
                            child: const Text('Cancelar'),
                          ),

                          const SizedBox(
                            width: AppSpacing.spacing8,
                          ),

                          FilledButton(
                            onPressed: guardando ? null : guardarCancion,
                            child: guardando
                                ? const SizedBox(
                                    width: 20,
                                    height: 20,
                                    child: CircularProgressIndicator(
                                      strokeWidth: 2,
                                    ),
                                  )
                                : const Text('Guardar'),
                          ),
                        ],
                      ),
                    ],
                  ),
                ),
              ),
            );
          },
        );
      },
    );
  } //acá termina el formulario y continúa la app

  @override
  Widget build(BuildContext context) {
    final colors = Theme.of(context).colorScheme;
    final text = Theme.of(context).textTheme;

    return Scaffold(
      // app bar
      appBar: AppBar(
        title: const Text('Listas'),
      ),

      // lista de canciones desde firestore
      body: StreamBuilder<QuerySnapshot<Map<String, dynamic>>>(
        stream: canciones.snapshots(),
        builder: (context, snapshot) {
          // error de conexión
          if (snapshot.hasError) {
            return const Center(
              child: Text('Error al cargar las canciones'),
            );
          }

          // estado de carga
          if (!snapshot.hasData) {
            return const Center(
              child: CircularProgressIndicator(),
            );
          }

          // documentos de firestore
          final documentos = snapshot.data!.docs;

          // lista vacía
          if (documentos.isEmpty) {
            return const Center(
              child: Text('No hay canciones todavía'),
            );
          }

          // construir lista dinámica
          return ListView.builder(
            padding: const EdgeInsets.symmetric(
              vertical: AppSpacing.spacing8,
            ),
            itemCount: documentos.length,
            itemBuilder: (context, index) {
              final datos = documentos[index].data();

              final titulo = datos['titulo'] as String? ?? '';
              final subtitulo = datos['subtitulo'] as String? ?? '';

              return ListTile(
                contentPadding: const EdgeInsets.symmetric(
                  horizontal: AppSpacing.spacing16,
                  vertical: AppSpacing.spacing4,
                ),

                // imagen local
                leading: ClipRRect(
                  borderRadius: BorderRadius.circular(
                    AppRadius.radius200,
                  ),
                  child: Image.asset(
                    'assets/images/kaleko.jpeg',
                    width: AppSpacing.spacing56,
                    height: AppSpacing.spacing56,
                    fit: BoxFit.cover,
                  ),
                ),

                // título
                title: Text(
                  titulo,
                  style: text.bodyLarge,
                ),

                // subtítulo
                subtitle: Text(
                  subtitulo,
                  style: text.bodyMedium?.copyWith(
                    color: colors.onSurfaceVariant,
                  ),
                ),

                // menú de acciones
                trailing: IconButton(
                  icon: const Icon(Icons.more_vert),
                  onPressed: () {},
                ),
              );
            },
          );
        },
      ),

      // botón para agregar canción
      floatingActionButton: FloatingActionButton(
        onPressed: agregarCancion, // trigger para la función declarada arriba
        child: const Icon(Icons.add),
      ),
    );
  }
}
